cmake_minimum_required(VERSION 3.18)
project(sapphire_package_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT SAPPHIRE_INSTALL_PREFIX)
  message(FATAL_ERROR "Set SAPPHIRE_INSTALL_PREFIX to a fresh core installation")
endif()
find_path(SAPPHIRE_INCLUDE_DIR slam.hpp PATHS "${SAPPHIRE_INSTALL_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_library(SAPPHIRE_LIBRARY NAMES sapphire_core.so PATHS "${SAPPHIRE_INSTALL_PREFIX}/lib" "${SAPPHIRE_INSTALL_PREFIX}/lib64" NO_DEFAULT_PATH REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core)
find_package(CUDAToolkit REQUIRED)
find_package(spdlog REQUIRED)
add_executable(consumer core.cpp)
target_include_directories(consumer PRIVATE "${SAPPHIRE_INCLUDE_DIR}" "${SAPPHIRE_INCLUDE_DIR}/thirdparty/3d_bbs" ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(consumer PRIVATE EIGEN_DONT_PARALLELIZE)
target_link_libraries(consumer PRIVATE "${SAPPHIRE_LIBRARY}" Eigen3::Eigen ${OpenCV_LIBS} CUDA::cudart spdlog::spdlog)
enable_testing()
add_test(NAME installed_core COMMAND consumer "${SAPPHIRE_INSTALL_PREFIX}/share/sapphire_core/config")
